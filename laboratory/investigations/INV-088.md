### 4.5 Proposed Scope Validator (For Reference)

**New File**: `/workspace/project/kde/runtime/ecu/scope/__init__.py`

```python
class ScopeValidator:
    """Validates investigation scope is KDE-related."""
    
    KDE_KEYWORDS = {
        'runtime', 'engine', 'seed', 'ecu', 'laboratory',
        'experiment', 'investigation', 'governance', 'principle',
        'knowledge', 'evidence', 'bootstrap', 'guard'
    }
    
    def is_kde_related(self, topic: str) -> bool:
        """Check if topic is KDE-related."""
        # Extract keywords from topic
        topic_lower = topic.lower()
        words = set(topic_lower.split())
        
        # Check for KDE keywords
        kde_hits = words.intersection(self.KDE_KEYWORDS)
        
        # Check for KDE prefixes
        kde_prefixes = ['kde-', 'kde_', 'lab-', 'inv-', 'seed-']
        for prefix in kde_prefixes:
            if prefix in topic_lower:
                return True
        
        # If significant KDE keywords found
        return len(kde_hits) >= 2 or any(kde_hits)
```

### 4.5 Integration Point

**Add to RuntimeECU:**

```python
class RuntimeECU:
    def __init__(self, kde_root: str):

        # Scope Validator
        self.scope_validator = ScopeValidator()
    
    def validate_investigation_scope(self, topic: str) -> bool:
        """Validate investigation topic is KDE-related."""
        if not self.scope_validator.is_kde_related(topic):
            return False
```
